   169→async function ensureData(forceReload = false) {
   170→  if (!cache.loaded || forceReload) {
   171→    const [sR, eR] = await Promise.all([
   172→      fetch('data/shipments.json?t=' + Date.now()),
   173→      fetch('data/exportaciones.json?t=' + Date.now()),
   174→    ]);
   175→    const allShipments: IngresoLine[] = await sR.json();
   176→    const allExports: ExpRecord[] = await eR.json();
   177→
   178→    // Merge: imported Excel data takes priority if static JSON is empty
   179→    // Read from trazabilidad_dep_imported (Excel import in Depositos tab)
   180→    let importedShipments: IngresoLine[] = [];
   181→    try {
   182→      const depImpRaw = localStorage.getItem('trazabilidad_dep_imported');
   183→      if (depImpRaw) importedShipments = JSON.parse(depImpRaw);
   184→    } catch { /* ignore */ }
   185→
   186→    // If static JSON is empty but imported data exists, use imported data as base
   187→    let baseShipments = allShipments;
   188→    if (baseShipments.length === 0 && importedShipments.length > 0) {
   189→      baseShipments = importedShipments;
   190→    }
   191→
   192→    // Filter to Caliral-bound shipments
   193→    let caliralShipments = baseShipments.filter(s => (s.nombreEstablecimientoDestino || '').toLowerCase().includes('caliral'));
   194→
   195→    // Also include imported records that are not already in base (for non-caliral merge)
   196→    if (baseShipments === allShipments && importedShipments.length > 0) {
   197→      const existingIds = new Set(caliralShipments.map(s => s.id));
   198→      for (const nr of importedShipments) {
   199→        if (!existingIds.has(nr.id)) {
   200→          caliralShipments.push(nr);
   201→        }
   202→      }
   203→    }
   204→
   205→    // Also include new deposit records from Depositos page (manual creates)
   206→    try {
   207→      const depNewRaw = localStorage.getItem('trazabilidad_dep_new_records');
   208→      if (depNewRaw) {
   209→        const depNew: IngresoLine[] = JSON.parse(depNewRaw);
   210→        const existingIds = new Set(caliralShipments.map(s => s.id));
   211→        for (const nr of depNew) {
   212→          if (!existingIds.has(nr.id)) {
   213→            caliralShipments.push(nr);
   214→          }
   215→        }
   216→      }
   217→    } catch { /* ignore */ }
   218→
   219→    // Apply deposit edits from Depositos page
   220→    try {
   221→      const depEditsRaw = localStorage.getItem('trazabilidad_dep_edits');
   222→      if (depEditsRaw) {
   223→        const depEdits: Record<string, Partial<IngresoLine>> = JSON.parse(depEditsRaw);
   224→        caliralShipments = caliralShipments.map(s => depEdits[s.id] ? { ...s, ...depEdits[s.id] } : s);
   225→      }
   226→    } catch { /* ignore */ }
   227→
   228→    // Remove deleted deposits
   229→    try {
   230→      const depDelRaw = localStorage.getItem('trazabilidad_dep_deleted');
   231→      if (depDelRaw) {
   232→        const depDeleted: Set<string> = new Set(JSON.parse(depDelRaw));
   233→        caliralShipments = caliralShipments.filter(s => !depDeleted.has(s.id));
   234→      }
   235→    } catch { /* ignore */ }
   236→
   237→    cache.shipments = caliralShipments;
   238→    cache.exportsRaw = allExports;
   239→    cache.loaded = true;
   240→  }
   241→
   242→  // Build full export list: static JSON + imported Excel + new PDF uploads (always re-read localStorage)
   243→  const allExportsList = [...cache.exportsRaw];
   244→
   245→  // Read from trazabilidad_exp_imported (Excel import in Exportaciones tab)
   246→  try {
   247→    const expImpRaw = localStorage.getItem('trazabilidad_exp_imported');
   248→    if (expImpRaw) {
   249→      const expImported: ExpRecord[] = JSON.parse(expImpRaw);
   250→      // If static JSON is empty, use imported as base
   251→      if (allExportsList.length === 0) {
   252→        allExportsList.push(...expImported);
   253→      } else {
   254→        const existingIds = new Set(allExportsList.map(e => e.id));
   255→        for (const nr of expImported) {
   256→          if (!existingIds.has(nr.id)) allExportsList.push(nr);
   257→        }
   258→      }
   259→    }
   260→  } catch { /* ignore */ }
   261→
   262→  // Also include new records from PDF uploads
   263→  try {
   264→    const raw = localStorage.getItem('trazabilidad_new_records');
   265→    if (raw) {
   266→      const newRecs: ExpRecord[] = JSON.parse(raw);
   267→      const existingIds = new Set(allExportsList.map(e => e.id));
   268→      for (const nr of newRecs) {
   269→        if (nr.tipo === 'EXPORTACION' && !existingIds.has(nr.id)) {
   270→          allExportsList.push(nr);
   271→        }
   272→      }
   273→    }
   274→  } catch { /* ignore */ }
   275→
   276→  // Apply edits from Exportaciones page to ALL exports (including PDF uploads)
   277→  const expEdits = loadExpEdits();
   278→  cache.exports = applyExpEdits(allExportsList, expEdits);
   279→}

   281→function extractIngresoCotes(obs: string | null | undefined, exportCote: string): string[] {
   282→  if (!obs) return [];
   283→  const allP = obs.match(/P\d{4,8}/gi) || [];
   284→  const allB = obs.match(/B\d{4,8}/gi) || [];
   285→  const all = [...allP, ...allB];
   286→  const exportUpper = exportCote.toUpperCase();
   287→  return all.map(c => c.toUpperCase()).filter(c => c !== exportUpper);
   288→}

   290→function aggregateByCote(lines: IngresoLine[]): Map<string, IngresoAgg> {
   291→  const map = new Map<string, IngresoAgg>();
   292→  for (const l of lines) {
   293→    const cote = l.nroCote?.trim();
   294→    if (!cote) continue;
   295→    if (!map.has(cote)) {
   296→      map.set(cote, {
   297→        cote, tramite: l.nroTramite, fecha: l.fechaTramite,
   298→        producto: l.denominacionMercaderia, cortes: [],
   299→        pesoNeto: 0, pesoBruto: 0, envases: 0, lineCount: 0, lines: [],
   300→      });
   301→    }
   302→    const agg = map.get(cote)!;
   303→    agg.lines.push(l);
   304→    agg.lineCount++;
   305→    agg.pesoNeto += l.pesoNeto || 0;
   306→    agg.pesoBruto += l.pesoBruto || 0;
   307→    agg.envases += l.cantidadEnvases || 0;
   308→    if (!agg.cortes.includes(l.corte)) agg.cortes.push(l.corte);
   309→  }
   310→  return map;
   311→}

   349→// --- Inline row for SinCruce with quick COTE linking ---
   350→function SinCruceInlineRow({ row, ingresoMap, stockAggMap, edits, onSaved, onEditFull, onViewDetail, isEditedFlag }: {
   351→  row: SinCruceRow;
   352→  ingresoMap: Map<string, IngresoAgg>;
   353→  stockAggMap: Map<string, StockCodigoAgg>;
   354→  edits: EditsStore;
   355→  onSaved: (edits: EditsStore) => void;
   356→  onEditFull: () => void;
   357→  onViewDetail: () => void;
   358→  isEditedFlag: boolean;
   359→}) {
   360→  const [expanding, setExpanding] = useState(false);
   361→  const [newCote, setNewCote] = useState('');
   362→  const [newCajas, setNewCajas] = useState('');
   363→
   364→  const currentManualCotes = edits.exports[row.exp.id]?.manualCotes || [];
   365→
   366→  const existingCotesInCaliral = [...ingresoMap.keys(), ...stockAggMap.keys()].sort();
   367→
   368→  const handleQuickAdd = () => {
   369→    const cote = newCote.trim().toUpperCase();
   370→    if (!cote) return;
   371→    let cajas = parseInt(newCajas) || 0;
   372→    if (cajas <= 0 && ingresoMap.has(cote)) {
   373→      cajas = ingresoMap.get(cote)!.envases;
   374→    }
   375→    if (cajas <= 0) {
   376→      toast.error('Ingresá la cantidad de cajas');
   377→      return;
   378→    }
   379→    const existingIdx = currentManualCotes.findIndex(c => c.cote === cote);
   380→    let updated: ManualCoteLink[];
   381→    if (existingIdx >= 0) {
   382→      updated = currentManualCotes.map((c, i) => i === existingIdx ? { ...c, cajas } : c);
   383→      toast.success(`${cote} actualizado: ${cajas} cajas`);
   384→    } else {
   385→      updated = [...currentManualCotes, { cote, cajas }];
   386→      toast.success(`${cote} agregado`);
   387→    }
   388→    const newEdits: EditsStore = {
   389→      ...edits,
   390→      exports: {
   391→        ...edits.exports,
   392→        [row.exp.id]: { ...edits.exports[row.exp.id], manualCotes: updated },
   393→      },
   394→    };
   395→    onSaved(newEdits);
   396→    setNewCote('');
   397→    setNewCajas('');
   398→  };
   399→
   400→  const handleRemoveCote = (cote: string) => {
   401→    const updated = currentManualCotes.filter(c => c.cote !== cote);
   402→    const newEdits: EditsStore = { ...edits };
   403→    if (updated.length > 0) {
   404→      newEdits.exports = { ...newEdits.exports, [row.exp.id]: { ...newEdits.exports[row.exp.id], manualCotes: updated } };
   405→    } else {
   406→      const existing = { ...newEdits.exports[row.exp.id] };
   407→      delete existing.manualCotes;
   408→      if (Object.keys(existing).length > 0) {
   409→        newEdits.exports = { ...newEdits.exports, [row.exp.id]: existing };
   410→      } else {
   411→        const { [row.exp.id]: _, ...rest } = newEdits.exports;
   412→        newEdits.exports = rest;
   413→      }
   414→    }
   415→    onSaved(newEdits);
   416→  };
   417→
   418→  return (
   419→    <>
   420→      <tr className={`border-b hover:bg-amber-50/40 ${isEditedFlag ? 'bg-violet-50/30' : ''}`}>
   421→        <td className="px-3 py-2.5 text-xs font-mono font-medium text-amber-700"><button onClick={(e) => { e.stopPropagation(); onViewDetail(); }} className="hover:underline cursor-pointer">{row.exp.nroCote}</button></td>
   422→        <td className="px-3 py-2.5 text-xs font-mono">{row.exp.nroTramite}</td>
   423→        <td className="px-3 py-2.5 text-xs">{fd(row.exp.fechaTramite)}</td>
   424→        <td className="px-3 py-2.5 text-xs">{row.exp.paisDestino}</td>
   425→        <td className="px-3 py-2.5 text-xs hidden lg:table-cell max-w-[200px] truncate">{row.exp.denominacionMercaderia}</td>
   426→        <td className="px-3 py-2.5 text-xs text-right font-mono font-medium">{(row.exp.cantidadEnvases || 0).toLocaleString('es-UY')}</td>
   427→        <td className="px-3 py-2.5 text-xs text-right font-mono hidden md:table-cell">{(row.exp.pesoNeto || 0).toLocaleString('es-UY')}</td>
   428→        <td className="px-3 py-2.5 text-center">
   429→          {currentManualCotes.length > 0 ? (
   430→            <div className="flex flex-wrap gap-1 justify-center">
   431→              {currentManualCotes.map(mc => {
   432→                const found = ingresoMap.get(mc.cote);
   433→                return (
   434→                  <span key={mc.cote} className={`inline-flex items-center gap-0.5 text-[10px] font-mono px-1.5 py-0.5 rounded ${found ? 'bg-emerald-100 text-emerald-800' : 'bg-red-100 text-red-700'}`}>
   435→                    {mc.cote} <span className="opacity-60">({mc.cajas})</span>
   436→                    <button className="ml-0.5 hover:text-red-600" onClick={() => handleRemoveCote(mc.cote)}><X className="h-2.5 w-2.5" /></button>
   437→                  </span>
   438→                );
   439→              })}
   440→              <button className="text-[10px] text-blue-600 hover:text-blue-800 font-medium px-1" onClick={() => setExpanding(!expanding)}>
   441→                {expanding ? 'cerrar' : '+ mas'}
   442→              </button>
   443→            </div>
   444→          ) : (
   445→            <button
   446→              className="inline-flex items-center gap-1 text-[11px] font-semibold text-white bg-violet-600 hover:bg-violet-700 px-2.5 py-1 rounded-md transition-colors"
   447→              onClick={() => setExpanding(!expanding)}
   448→            >
   449→              <Plus className="h-3 w-3" />Vincular COTE
   450→            </button>
   451→          )}
   452→        </td>
   453→      </tr>
   454→      {expanding && (
   455→        <tr className="border-b bg-violet-50/50">
   456→          <td colSpan={8} className="px-3 py-3">
   457→            <div className="flex flex-wrap items-end gap-2">
   458→              <div className="flex-1 min-w-[140px]">
   459→                <FieldLabel>COTE de Ingreso</FieldLabel>
   460→                <Input
   461→                  placeholder="Ej: P12345"
   462→                  value={newCote}
   463→                  onChange={e => setNewCote(e.target.value.toUpperCase())}
   464→                  className="h-8 text-xs font-mono"
   465→                  list={`cote-sug-${row.exp.id}`}
   466→                  onKeyDown={e => { if (e.key === 'Enter') { e.preventDefault(); handleQuickAdd(); } }}
   467→                />
   468→                <datalist id={`cote-sug-${row.exp.id}`}>
   469→                  {existingCotesInCaliral.filter(c => !currentManualCotes.some(mc => mc.cote === c)).map(c => {
   470→                    const ing = ingresoMap.get(c);
   471→                    const stk = stockAggMap.get(c);
