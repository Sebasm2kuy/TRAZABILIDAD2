   169→   169→async function ensureData(forceReload = false) {
   170→   170→  if (!cache.loaded || forceReload) {
   171→   171→    const [sR, eR] = await Promise.all([
   172→   172→      fetch('data/shipments.json?t=' + Date.now()),
   173→   173→      fetch('data/exportaciones.json?t=' + Date.now()),
   174→   174→    ]);
   175→   175→    const allShipments: IngresoLine[] = await sR.json();
   176→   176→    const allExports: ExpRecord[] = await eR.json();
   177→   177→
   178→   178→    // Merge: imported Excel data takes priority if static JSON is empty
   179→   179→    // Read from trazabilidad_dep_imported (Excel import in Depositos tab)
   180→   180→    let importedShipments: IngresoLine[] = [];
   181→   181→    try {
   182→   182→      const depImpRaw = localStorage.getItem('trazabilidad_dep_imported');
   183→   183→      if (depImpRaw) importedShipments = JSON.parse(depImpRaw);
   184→   184→    } catch { /* ignore */ }
   185→   185→
   186→   186→    // If static JSON is empty but imported data exists, use imported data as base
   187→   187→    let baseShipments = allShipments;
   188→   188→    if (baseShipments.length === 0 && importedShipments.length > 0) {
   189→   189→      baseShipments = importedShipments;
   190→   190→    }
   191→   191→
   192→   192→    // Filter to Caliral-bound shipments
   193→   193→    let caliralShipments = baseShipments.filter(s => (s.nombreEstablecimientoDestino || '').toLowerCase().includes('caliral'));
   194→   194→
   195→   195→    // Also include imported records that are not already in base (for non-caliral merge)
   196→   196→    if (baseShipments === allShipments && importedShipments.length > 0) {
   197→   197→      const existingIds = new Set(caliralShipments.map(s => s.id));
   198→   198→      for (const nr of importedShipments) {
   199→   199→        if (!existingIds.has(nr.id)) {
   200→   200→          caliralShipments.push(nr);
   201→   201→        }
   202→   202→      }
   203→   203→    }
   204→   204→
   205→   205→    // Also include new deposit records from Depositos page (manual creates)
   206→   206→    try {
   207→   207→      const depNewRaw = localStorage.getItem('trazabilidad_dep_new_records');
   208→   208→      if (depNewRaw) {
   209→   209→        const depNew: IngresoLine[] = JSON.parse(depNewRaw);
   210→   210→        const existingIds = new Set(caliralShipments.map(s => s.id));
   211→   211→        for (const nr of depNew) {
   212→   212→          if (!existingIds.has(nr.id)) {
   213→   213→            caliralShipments.push(nr);
   214→   214→          }
   215→   215→        }
   216→   216→      }
   217→   217→    } catch { /* ignore */ }
   218→   218→
   219→   219→    // Apply deposit edits from Depositos page
   220→   220→    try {
   221→   221→      const depEditsRaw = localStorage.getItem('trazabilidad_dep_edits');
   222→   222→      if (depEditsRaw) {
   223→   223→        const depEdits: Record<string, Partial<IngresoLine>> = JSON.parse(depEditsRaw);
   224→   224→        caliralShipments = caliralShipments.map(s => depEdits[s.id] ? { ...s, ...depEdits[s.id] } : s);
   225→   225→      }
   226→   226→    } catch { /* ignore */ }
   227→   227→
   228→   228→    // Remove deleted deposits
   229→   229→    try {
   230→   230→      const depDelRaw = localStorage.getItem('trazabilidad_dep_deleted');
   231→   231→      if (depDelRaw) {
   232→   232→        const depDeleted: Set<string> = new Set(JSON.parse(depDelRaw));
   233→   233→        caliralShipments = caliralShipments.filter(s => !depDeleted.has(s.id));
   234→   234→      }
   235→   235→    } catch { /* ignore */ }
   236→   236→
   237→   237→    cache.shipments = caliralShipments;
   238→   238→    cache.exportsRaw = allExports;
   239→   239→    cache.loaded = true;
   240→   240→  }
   241→   241→
   242→   242→  // Build full export list: static JSON + imported Excel + new PDF uploads (always re-read localStorage)
   243→   243→  const allExportsList = [...cache.exportsRaw];
   244→   244→
   245→   245→  // Read from trazabilidad_exp_imported (Excel import in Exportaciones tab)
   246→   246→  try {
   247→   247→    const expImpRaw = localStorage.getItem('trazabilidad_exp_imported');
   248→   248→    if (expImpRaw) {
   249→   249→      const expImported: ExpRecord[] = JSON.parse(expImpRaw);
   250→   250→      // If static JSON is empty, use imported as base
   251→   251→      if (allExportsList.length === 0) {
   252→   252→        allExportsList.push(...expImported);
   253→   253→      } else {
   254→   254→        const existingIds = new Set(allExportsList.map(e => e.id));
   255→   255→        for (const nr of expImported) {
   256→   256→          if (!existingIds.has(nr.id)) allExportsList.push(nr);
   257→   257→        }
   258→   258→      }
   259→   259→    }
   260→   260→  } catch { /* ignore */ }
   261→   261→
   262→   262→  // Also include new records from PDF uploads
   263→   263→  try {
   264→   264→    const raw = localStorage.getItem('trazabilidad_new_records');
   265→   265→    if (raw) {
   266→   266→      const newRecs: ExpRecord[] = JSON.parse(raw);
   267→   267→      const existingIds = new Set(allExportsList.map(e => e.id));
   268→   268→      for (const nr of newRecs) {
   269→   269→        if (nr.tipo === 'EXPORTACION' && !existingIds.has(nr.id)) {
   270→   270→          allExportsList.push(nr);
   271→   271→        }
   272→   272→      }
   273→   273→    }
   274→   274→  } catch { /* ignore */ }
   275→   275→
   276→   276→  // Apply edits from Exportaciones page to ALL exports (including PDF uploads)
   277→   277→  const expEdits = loadExpEdits();
   278→   278→  cache.exports = applyExpEdits(allExportsList, expEdits);
   279→   279→}

   281→   281→function extractIngresoCotes(obs: string | null | undefined, exportCote: string): string[] {
   282→   282→  if (!obs) return [];
   283→   283→  const allP = obs.match(/P\d{4,8}/gi) || [];
   284→   284→  const allB = obs.match(/B\d{4,8}/gi) || [];
   285→   285→  const all = [...allP, ...allB];
   286→   286→  const exportUpper = exportCote.toUpperCase();
   287→   287→  return all.map(c => c.toUpperCase()).filter(c => c !== exportUpper);
   288→   288→}

   290→   290→function aggregateByCote(lines: IngresoLine[]): Map<string, IngresoAgg> {
   291→   291→  const map = new Map<string, IngresoAgg>();
   292→   292→  for (const l of lines) {
   293→   293→    const cote = l.nroCote?.trim();
   294→   294→    if (!cote) continue;
   295→   295→    if (!map.has(cote)) {
   296→   296→      map.set(cote, {
   297→   297→        cote, tramite: l.nroTramite, fecha: l.fechaTramite,
   298→   298→        producto: l.denominacionMercaderia, cortes: [],
   299→   299→        pesoNeto: 0, pesoBruto: 0, envases: 0, lineCount: 0, lines: [],
   300→   300→      });
   301→   301→    }
   302→   302→    const agg = map.get(cote)!;
   303→   303→    agg.lines.push(l);
   304→   304→    agg.lineCount++;
   305→   305→    agg.pesoNeto += l.pesoNeto || 0;
   306→   306→    agg.pesoBruto += l.pesoBruto || 0;
   307→   307→    agg.envases += l.cantidadEnvases || 0;
   308→   308→    if (!agg.cortes.includes(l.corte)) agg.cortes.push(l.corte);
   309→   309→  }
   310→   310→  return map;
   311→   311→}

   350→   350→function SinCruceInlineRow({ row, ingresoMap, stockAggMap, edits, onSaved, onEditFull, onViewDetail, isEditedFlag }: {
   351→   351→  row: SinCruceRow;
   352→   352→  ingresoMap: Map<string, IngresoAgg>;
   353→   353→  stockAggMap: Map<string, StockCodigoAgg>;
   354→   354→  edits: EditsStore;
   355→   355→  onSaved: (edits: EditsStore) => void;
   356→   356→  onEditFull: () => void;
   357→   357→  onViewDetail: () => void;
   358→   358→  isEditedFlag: boolean;
   359→   359→}) {
   360→   360→  const [expanding, setExpanding] = useState(false);
   361→   361→  const [newCote, setNewCote] = useState('');
   362→   362→  const [newCajas, setNewCajas] = useState('');
   363→   363→
   364→   364→  const currentManualCotes = edits.exports[row.exp.id]?.manualCotes || [];
   365→   365→
   366→   366→  const existingCotesInCaliral = [...ingresoMap.keys(), ...stockAggMap.keys()].sort();
   367→   367→
   368→   368→  const handleQuickAdd = () => {
   369→   369→    const cote = newCote.trim().toUpperCase();
   370→   370→    if (!cote) return;
   371→   371→    let cajas = parseInt(newCajas) || 0;
   372→   372→    if (cajas <= 0 && ingresoMap.has(cote)) {
   373→   373→      cajas = ingresoMap.get(cote)!.envases;
   374→   374→    }
   375→   375→    if (cajas <= 0) {
   376→   376→      toast.error('Ingresá la cantidad de cajas');
   377→   377→      return;
   378→   378→    }
   379→   379→    const existingIdx = currentManualCotes.findIndex(c => c.cote === cote);
   380→   380→    let updated: ManualCoteLink[];
   381→   381→    if (existingIdx >= 0) {
   382→   382→      updated = currentManualCotes.map((c, i) => i === existingIdx ? { ...c, cajas } : c);
   383→   383→      toast.success(`${cote} actualizado: ${cajas} cajas`);
   384→   384→    } else {
   385→   385→      updated = [...currentManualCotes, { cote, cajas }];
   386→   386→      toast.success(`${cote} agregado`);
   387→   387→    }
   388→   388→    const newEdits: EditsStore = {
   389→   389→      ...edits,
   390→   390→      exports: {
   391→   391→        ...edits.exports,
   392→   392→        [row.exp.id]: { ...edits.exports[row.exp.id], manualCotes: updated },
   393→   393→      },
   394→   394→    };
   395→   395→    onSaved(newEdits);
   396→   396→    setNewCote('');
   397→   397→    setNewCajas('');
   398→   398→  };
   399→   399→
   400→   400→  const handleRemoveCote = (cote: string) => {
   401→   401→    const updated = currentManualCotes.filter(c => c.cote !== cote);
   402→   402→    const newEdits: EditsStore = { ...edits };
   403→   403→    if (updated.length > 0) {
   404→   404→      newEdits.exports = { ...newEdits.exports, [row.exp.id]: { ...newEdits.exports[row.exp.id], manualCotes: updated } };
   405→   405→    } else {
   406→   406→      const existing = { ...newEdits.exports[row.exp.id] };
   407→   407→      delete existing.manualCotes;
   408→   408→      if (Object.keys(existing).length > 0) {
   409→   409→        newEdits.exports = { ...newEdits.exports, [row.exp.id]: existing };
   410→   410→      } else {
   411→   411→        const { [row.exp.id]: _, ...rest } = newEdits.exports;
   412→   412→        newEdits.exports = rest;
   413→   413→      }
   414→   414→    }
   415→   415→    onSaved(newEdits);
   416→   416→  };
   417→   417→
   418→   418→  return (
   419→   419→    <>

   523→   523→function StockTable({ stockAggMap, ingresoMap, cruceRows, sinCruceRows, edits }: {

   535→   535→  // Manual links: use specified cajas per COTE
   536→   536→  // Automatic links: distribute export cajas proportionally by ingreso cajas

   538→   538→    const map = new Map<string, number>();
   539→   539→    // From cruceRows (exports matched to ingreso COTEs)

   541→   541→      if (r.isManualLink && edits.exports[r.exp.id]?.manualCotes) {
   542→   542→        // Manual: use exact cajas per COTE

   544→   544→          map.set(mc.cote, (map.get(mc.cote) || 0) + mc.cajas);
   545→   545→        }
   546→   546→      } else if (r.ingresoCotes.length > 0) {
   547→   547→        // Automatic: distribute export cajas proportionally by ingreso envases
   548→   548→        const totalIngCajas = r.ingresoCotes.reduce((s, c) => s + (ingresoMap.get(c)?.envases || 0), 0);

   601→   601→    items.sort((a, b) => b.totalKilos - a.totalKilos);
